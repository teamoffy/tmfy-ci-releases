# Small public binaries, mirrored verified as tar.zst (one release per tool:
# <name>/v<version>). "latest" = the GitHub repo's latest release tag.
#
# columns:
#   name          product/tool name ([a-z0-9-]+)
#   gh-repo       owner/repo whose latest release is the version source;
#                 nodejs/node is special-cased to the newest LTS line (see
#                 tool_latest in check.sh — its GH latest is the Current line)
#   url-x64       download URL for the linux-x64 asset ("-" to skip that arch)
#   url-arm64     same for linux-arm64
#   mode          bin              — URL is the bare binary -> usr/local/bin/<name>
#                 tarbin:<member>  — tar.* archive; the file named <member> ->
#                                  usr/local/bin/<member>
#                 tardir           — tar.* archive is the whole install ->
#                                  opt/<name>/
#
# URLs expand {tag} = upstream tag, {ver} = tag without a leading v.
# Verification: github.com release URLs are checked against the asset's GitHub
# digest; other hosts need a <url>.sha256/.sha256sum sidecar or a per-directory
# SHASUMS256.txt/sha256sums.txt covering the asset.
#
